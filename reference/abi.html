    uint32_t size, version;
    void *user;
    const char *id;

    // Copy request strings before request returns. The sink table remains valid
    // until provider shutdown returns; cancelled request IDs cannot publish.
    int (*request)(void *, const LyraMediaRequestV3 *, const LyraMediaSinkV3 *);
    void (*cancel)(void *, uint64_t);
} LyraMediaProviderV3;

typedef struct LyraMediaContextV3 {

    int (*provider_id)(void *, int, char *, int);

    int (*query_text)(void *, char *, int);
    // Compare-and-set; rejects an asynchronous response to an older query.
    int (*set_query)(void *, uint64_t, const char *);
    uint64_t (*request)(void *, const char *provider, uint32_t kind,
                        const char *collection, const char *query, const char *cursor, const char *uid);
    void (*cancel)(void *, uint64_t);
    uint64_t (*revision)(void *, uint64_t);
    int (*count)(void *, uint64_t);
    // Output strings are borrowed until the next frame boundary/revision.
    int (*item)(void *, uint64_t, int, LyraMediaDescriptorV3 *);
    int (*select)(void *, uint64_t, int);
    // 1 ready, 0 pending, -1 invalid. Ready transfers fd; copy strings before cancel.
    int (*take_resource)(void *, uint64_t, LyraMediaResourceV3 *);
    int (*current)(void *, LyraMediaDescriptorV3 *);
    int (*queue_count)(void *);
    int (*queue_item)(void *, int, LyraMediaDescriptorV3 *);

#pragma once
#include <stdint.h>
// One current protocol. Optional fields extend messages, not C layouts.
#define LYRA_FIELD_REQUIRED 1u
#define LYRA_FIELD_I64 1u
#define LYRA_FIELD_F64 2u
#define LYRA_FIELD_UTF8 3u
#define LYRA_FIELD_BYTES 4u
#define LYRA_FIELD_HANDLE 5u
#define LYRA_MESSAGE_OK 0
#define LYRA_MESSAGE_INVALID -1
#define LYRA_MESSAGE_UNSUPPORTED -2
#define LYRA_MESSAGE_CAPACITY -3

typedef struct LyraFieldV3 {
    uint32_t tag, type, flags, bytes;
    const void *data;
} LyraFieldV3;
typedef struct LyraMessageV3 {
    uint32_t size, field_count;
    uint64_t operation;
    const LyraFieldV3 *fields;
} LyraMessageV3;
// Caller owns field descriptors; returned data is borrowed until the next call.
typedef struct LyraReplyV3 {
    uint32_t size, capacity, field_count;
    int32_t status;
    LyraFieldV3 *fields;
} LyraReplyV3;
typedef struct LyraCapabilityV3 {
    uint32_t size;
    const char *name;
    void *user;
    int (*request)(void *, const LyraMessageV3 *, LyraReplyV3 *);
} LyraCapabilityV3;
// Endpoint-owned schemas validate known fields. Unknown optional fields are
// ignored; unknown required fields fail. No older wire layout is accepted.
typedef struct LyraFieldRuleV3 { uint32_t tag, type; } LyraFieldRuleV3;
static inline int lyra_message_fields(const LyraMessageV3 *message,
                                      const LyraFieldRuleV3 *rules, uint32_t count) {
    if (!message || message->size != sizeof(*message) ||
        (message->field_count && !message->fields) || (count && !rules)) return LYRA_MESSAGE_INVALID;
    for (uint32_t i = 0; i < message->field_count; ++i) {
        const LyraFieldV3 *field = &message->fields[i];
        uint32_t r = 0;
        for (; r < count && rules[r].tag != field->tag; ++r) {}
        if (r == count) { if (field->flags & LYRA_FIELD_REQUIRED) return LYRA_MESSAGE_UNSUPPORTED; }
        else if (rules[r].type != field->type) return LYRA_MESSAGE_INVALID;
    }
    return LYRA_MESSAGE_OK;
}

#pragma once
#include <stdint.h>
#include "message_contracts.h"
// Registry-local, opaque capability identities, not pointers and not persistent IDs.
// Thread: lookup/enumeration/mutation/execution run on the registry owner
// (creating) thread, which is the host frame thread.
// 0 means absent/invalid. Nonzero handles are never reused in this registry instance.
// A default capability keeps its handle through provide(same name), refresh,
// external replacement and restoration. RuntimeServices caches only these defaults.
// A capability with no default is revoked when its module/publisher unloads;
// replacing that module or publishing the same name again obtains a NEW handle.
// Refresh of unchanged modules preserves handles; refresh that retires a module
// follows the above revocation rule. Enumeration order/index is not stable.
// On stale/0/detached handles: describe/frame/input/supply return -1,
// interface returns NULL; request returns UNSUPPORTED (detached: INVALID).
// describe requires size==sizeof; for a correctly sized output, any failure
// clears flags/handle/string pointers. Invalid-size outputs are untouched.
// describe strings and interface pointers are borrowed until registry mutation;
// do not retain them across refresh/unload. Re-resolve revoked identities by name.
// Context pointers are borrowed while their registry is alive; detached error
// behavior does not permit calls after registry destruction.
// Handles from another instance must never be passed here. No private C++ objects
// cross this ABI.
#define LYRA_INTERFACE_SERVICE_CONTEXT 16u
#define LYRA_SERVICE_SELECTED 0u
#define LYRA_SERVICE_DEFAULT 1u
#define LYRA_SERVICE_OVERRIDDEN 1u
#define LYRA_SERVICE_FRAME 2u
#define LYRA_SERVICE_INPUT 4u
#define LYRA_SERVICE_SUPPLY 16u
#define LYRA_SERVICE_FALLBACK_FRAME 32u
#define LYRA_SERVICE_DEFAULT_ACTIVE 64u
#define LYRA_SERVICE_REQUEST 128u
struct LyraInput;
struct LyraBox;
typedef struct LyraServiceStateV3 {
  uint32_t size, flags;

  uint32_t size, version;

  uint64_t (*publish)(void *, const LyraCapabilityV3 *);
  int (*request)(void *, uint64_t, const LyraMessageV3 *, LyraReplyV3 *);
  int (*describe)(void *, uint64_t, LyraServiceStateV3 *);
  int (*frame)(void *, uint64_t, uint32_t implementation, int64_t,
               const struct LyraBox *);
  int (*input)(void *, uint64_t, uint32_t, const struct LyraInput *);
  int (*hit)(void *, const struct LyraInput *);
  int (*supply)(void *, uint64_t, uint32_t, int64_t, char *, int);
